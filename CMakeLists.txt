cmake_minimum_required(VERSION 3.22)
project(DSMEngine LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "DSMEngine 当前仅支持 Windows/MSVC")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

option(DSM_BUILD_PROJECTS "构建仓库内示例项目" ON)
option(DSM_BUILD_EDITOR "构建编辑器" ON)
option(DSM_BUILD_PBR "构建 PBR 项目" ON)
option(DSM_BUILD_RAYTRACING "构建 RayTracing 项目" ON)
option(DSM_BUILD_VALIDATION "构建渲染验证入口" ON)
set(DSM_PROJECT_FILE "" CACHE FILEPATH "作为构建入口的 .dsmproj 文件")

if(DSM_PROJECT_FILE)
    get_filename_component(DSM_PROJECT_FILE "${DSM_PROJECT_FILE}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
    if(NOT EXISTS "${DSM_PROJECT_FILE}")
        message(FATAL_ERROR "项目描述文件不存在: ${DSM_PROJECT_FILE}")
    endif()
    get_filename_component(DSM_PROJECT_ROOT "${DSM_PROJECT_FILE}" DIRECTORY)
    get_filename_component(DSM_PROJECT_NAME "${DSM_PROJECT_FILE}" NAME_WE)
    if(DSM_PROJECT_NAME STREQUAL "PBR")
        set(DSM_BUILD_PROJECTS ON)
        set(DSM_BUILD_PBR ON)
        set(DSM_BUILD_RAYTRACING OFF)
        set(DSM_BUILD_EDITOR ON)
    elseif(DSM_PROJECT_NAME STREQUAL "RayTracing")
        set(DSM_BUILD_PROJECTS ON)
        set(DSM_BUILD_PBR OFF)
        set(DSM_BUILD_RAYTRACING ON)
        set(DSM_BUILD_EDITOR ON)
    else()
        message(FATAL_ERROR "没有与项目描述文件匹配的构建目标: ${DSM_PROJECT_NAME}")
    endif()
endif()

add_compile_definitions(UNICODE _UNICODE SPDLOG_WCHAR_TO_UTF8_SUPPORT SPDLOG_UTF8_TO_WCHAR_CONSOLE)

# 将与 MSVC 目标 SDK 配套的 DXC DLL 部署到可执行目录。
function(dsm_stage_dxc target_name)
    get_filename_component(system_programs_root "$ENV{ProgramFiles}" DIRECTORY)
    set(sdk_bin "${system_programs_root}/Program Files (x86)/Windows Kits/10/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64")
    if(NOT EXISTS "${sdk_bin}/dxcompiler.dll" OR NOT EXISTS "${sdk_bin}/dxil.dll")
        message(FATAL_ERROR "未找到 Windows SDK DXC 运行库: ${sdk_bin}")
    endif()
    add_custom_command(TARGET ${target_name} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${sdk_bin}/dxcompiler.dll" "${sdk_bin}/dxil.dll" "$<TARGET_FILE_DIR:${target_name}>"
        VERBATIM)
endfunction()

    if(DSM_BUILD_RAYTRACING)
        add_subdirectory(Projects/RayTracing)
    endif()
endif()
